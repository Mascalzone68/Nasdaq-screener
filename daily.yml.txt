name: NASDAQ 3 Verdi Daily
on:
  schedule:
    - cron: '30 20 * * 1-5'
  workflow_dispatch:
jobs:
  screener:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - run: pip install finviz yfinance pandas lxml
      - run: python nasdaq_3verdi_puro.py
      - uses: actions/upload-artifact@v4
        with:
          name: nasdaq-list
          path: "*.csv"
